========================================
CLIENTE-VPN-3
========================================

# Ver direccionamiento
ip addr show eth0

# Ver tabla de rutas
ip route

# Configuración temporal en VLAN 10
ip addr flush dev eth0
ip addr add 10.12.48.11/25 dev eth0
ip link set eth0 up
ip route add default via 10.12.48.1

# Configurar DNS
echo "nameserver 8.8.8.8" > /etc/resolv.conf

# Pruebas de conectividad
ping -c 4 10.12.48.1
ping -c 4 203.0.113.3
ping -c 4 8.8.8.8
ping -c 4 deb.debian.org

# Instalar StrongSwan
apt update
apt install -y strongswan-swanctl charon-systemd
apt install -y strongswan strongswan-starter

# Arrancar StrongSwan
ipsec start

# Ver estado de StrongSwan
ipsec statusall

# Ver algoritmos disponibles
ipsec listalgs | grep -i des
ipsec listalgs | grep -Ei 'SHA1|SHA256|ECP_384|ECP_521|ECP'

# Verificar parámetros IKE/ESP cargados
grep -E '^[[:space:]]*(ike|esp)=' /etc/ipsec.conf

# Reiniciar StrongSwan
ipsec restart

# Levantar VPN
ipsec up VPN_REMOTA_3

# Verificar VPN activa
ipsec statusall

# Prueba SSH sin VPN
ssh -o ConnectTimeout=5 gerson@10.12.48.130

# Eliminar clave SSH antigua
ssh-keygen -f "/root/.ssh/known_hosts" -R "10.12.48.130"

# Prueba SSH con VPN
ssh gerson@10.12.48.130


========================================
WEB-SERVER-3
========================================

# Configurar IP del servidor
ip addr add 10.12.48.130/28 dev eth0
ip link set eth0 up
ip route add default via 10.12.48.129

# Ver direccionamiento y rutas
ip addr show eth0
ip route

# Configurar DNS
echo "nameserver 8.8.8.8" > /etc/resolv.conf

# Probar conectividad
ping -c 4 10.12.48.129
ping -c 4 8.8.8.8
ping -c 4 archive.ubuntu.com

# Verificar servicio HTTPS
ss -lntp | grep :443

# Validar e iniciar Nginx
nginx -t
service nginx start

# Instalar SSH
apt update
apt install -y openssh-server

# Iniciar SSH
mkdir -p /run/sshd
service ssh start

# Verificar puerto SSH
ss -lntp | grep :22

# Crear usuario para acceso SSH
adduser gerson

# Verificar usuario
id gerson
